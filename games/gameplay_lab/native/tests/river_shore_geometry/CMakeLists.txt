cmake_minimum_required(VERSION 3.24)
project(RiverShoreGeometryCpu LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
file(READ "${ROOT}/games/gameplay_lab/native/src/GameplayLabGame.cpp" GAME_SOURCE)
string(FIND "${GAME_SOURCE}" "void CarveShowcaseRiverShore3D(" FUNCTION_BEGIN)
string(FIND "${GAME_SOURCE}" "Mesh3D CreateCottageFacadeMesh3D(" FUNCTION_END)
math(EXPR FUNCTION_LENGTH "${FUNCTION_END}-${FUNCTION_BEGIN}")
string(SUBSTRING "${GAME_SOURCE}" ${FUNCTION_BEGIN} ${FUNCTION_LENGTH} RIVER_FUNCTION)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/RiverProductionExtract.h" "${RIVER_FUNCTION}")
add_executable(RiverShoreGeometryCpu RiverShoreGeometryCpu.cpp "${ROOT}/engine/native/src/HeightField3D.cpp")
target_compile_features(RiverShoreGeometryCpu PRIVATE cxx_std_20)
target_include_directories(RiverShoreGeometryCpu PRIVATE "${ROOT}/engine/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(RiverShoreGeometryCpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(RiverShoreGeometryCpu PRIVATE /utf-8 /W4 /permissive-)
add_test(NAME RiverShoreGeometryCpu COMMAND RiverShoreGeometryCpu "${CMAKE_CURRENT_BINARY_DIR}")
