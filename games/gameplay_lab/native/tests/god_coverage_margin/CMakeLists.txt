cmake_minimum_required(VERSION 3.24)
project(GodCoverageMarginCpu LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(LAB "${ROOT}/games/gameplay_lab/native")
add_executable(GodCoverageMarginCpu GodCoverageMarginCpu.cpp)
target_compile_features(GodCoverageMarginCpu PRIVATE cxx_std_20)
target_include_directories(GodCoverageMarginCpu PRIVATE "${LAB}/include" "${ROOT}/engine/native/include")
target_compile_definitions(GodCoverageMarginCpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE AEVRYND_CHARACTER_BUILD_FINGERPRINT="god-margin-cpu")
target_compile_options(GodCoverageMarginCpu PRIVATE /utf-8 /W4 /permissive-)
add_test(NAME GodCoverageMarginCpu COMMAND GodCoverageMarginCpu "${CMAKE_CURRENT_BINARY_DIR}/coverage-margin-receipt.json")
set(CPU_ENGINE_LIBRARY "" CACHE FILEPATH "Existing engine archive for CPU integration regression")
if(CPU_ENGINE_LIBRARY)
  add_executable(GodCoverageProductionContracts "${LAB}/tests/GameplayCameraProductionIntegrationSmoke.cpp" "${LAB}/src/GameplayFirstPersonViewmodel3D.cpp" "${LAB}/src/GameplayAnimation3D.cpp" "${LAB}/src/GameplayCameraOcclusionPolicy3D.cpp" "${LAB}/src/GameplayCameraProductionBuilder3D.cpp" "${LAB}/src/GameplayOccluderMask3D.cpp" "${LAB}/src/GameplayShowcaseEnvironment3D.cpp" "${LAB}/src/ShowcaseLandEcology3D.cpp")
  target_compile_features(GodCoverageProductionContracts PRIVATE cxx_std_20)
  target_include_directories(GodCoverageProductionContracts PRIVATE "${LAB}/include" "${ROOT}/engine/native/include")
  target_compile_definitions(GodCoverageProductionContracts PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE AEVRYND_SOURCE_DIR=L"${ROOT}")
  target_compile_options(GodCoverageProductionContracts PRIVATE /utf-8 /W4 /permissive-)
  target_link_libraries(GodCoverageProductionContracts PRIVATE "${CPU_ENGINE_LIBRARY}" d3d12 dxgi windowscodecs ole32 user32 gdi32 psapi xaudio2 xinput9_1_0 bcrypt)
  add_test(NAME GodCoverageProductionContracts COMMAND GodCoverageProductionContracts)
endif()
