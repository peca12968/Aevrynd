cmake_minimum_required(VERSION 3.24)
project(GodFrozenPoseCpu LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(LAB "${ROOT}/games/gameplay_lab/native")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LAB}/src/GameplayLabGame.cpp")
file(READ "${LAB}/src/GameplayLabGame.cpp" SOURCE)
string(REGEX MATCH "const SkinnedModel3D& godSubjectModel=[^;]+;([^#]+)if\\(!SkinModelInto\\(godSubjectModel" BLOCK "${SOURCE}")
set(SELECTOR "${CMAKE_MATCH_1}")
if(NOT SELECTOR)
  message(FATAL_ERROR "God authority pose selector missing")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AuthoritySelector.inc" "${SELECTOR}\nreturn HashGameplayCapturePose3D(godSubjectPose);\n")
string(REGEX MATCH "bool GameplayGodPoseAuthorityPass3D\\([^}]+}" AUTHORITY_GATE "${SOURCE}")
if(NOT AUTHORITY_GATE)
  message(FATAL_ERROR "God authority phase gate missing")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AuthorityGate.inc" "${AUTHORITY_GATE}\n")
add_executable(GodFrozenPoseCpu GodFrozenPoseCpu.cpp)
target_compile_features(GodFrozenPoseCpu PRIVATE cxx_std_20)
target_include_directories(GodFrozenPoseCpu PRIVATE "${LAB}/include" "${ROOT}/engine/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(GodFrozenPoseCpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN AEVRYND_CHARACTER_BUILD_FINGERPRINT="god-frozen-cpu")
target_compile_options(GodFrozenPoseCpu PRIVATE /utf-8 /W4)
add_test(NAME GodFrozenPoseCpu COMMAND GodFrozenPoseCpu)
