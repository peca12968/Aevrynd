include("${CMAKE_CURRENT_LIST_DIR}/../ReplicationSources.cmake")
add_library(AevryndReplicationCore STATIC ${AEVRYND_REPLICATION_SOURCES})
target_include_directories(AevryndReplicationCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_compile_features(AevryndReplicationCore PUBLIC cxx_std_20)
target_compile_definitions(AevryndReplicationCore PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AevryndReplicationCore PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(AevryndReplicationCore PUBLIC bcrypt)

foreach(suite ReplicationProtocol3D ReplicationRuntime3D)
  add_executable(Aevrynd${suite}Test "../tests/${suite}Smoke.cpp")
  target_compile_options(Aevrynd${suite}Test PRIVATE /W4 /permissive- /utf-8)
  target_link_libraries(Aevrynd${suite}Test PRIVATE AevryndReplicationCore)
  add_test(NAME Aevrynd${suite}Test COMMAND Aevrynd${suite}Test)
endforeach()
