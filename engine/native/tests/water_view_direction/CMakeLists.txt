cmake_minimum_required(VERSION 3.24)
project(WaterViewDirection LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(SHADER "${ROOT}/engine/native/shaders/World.hlsl")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHADER}")
file(READ "${SHADER}" SOURCE)
string(REGEX MATCH "float3 WaterSurfaceViewDirection\\([^}]+}" FUNCTION "${SOURCE}")
if(NOT FUNCTION)
 message(FATAL_ERROR "Production water direction helper missing")
endif()
string(REPLACE "float4x4" "Matrix4" FUNCTION "${FUNCTION}")
string(REPLACE "float3" "Vec3" FUNCTION "${FUNCTION}")
string(REPLACE "normalize(" "Normalize(" FUNCTION "${FUNCTION}")
string(REPLACE "abs(" "std::abs(" FUNCTION "${FUNCTION}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WaterDirection.inc" "${FUNCTION}\n")
foreach(target IN ITEMS WaterViewDirectionCpu WaterOpticsCpu)
 if(target STREQUAL "WaterViewDirectionCpu")
  set(testSource "${CMAKE_CURRENT_SOURCE_DIR}/WaterViewDirectionCpu.cpp")
 else()
  set(testSource "${ROOT}/engine/native/tests/WaterOptics3DSmoke.cpp")
 endif()
 add_executable(${target} "${testSource}")
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_options(${target} PRIVATE /utf-8 /W4)
 target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_include_directories(${target} PRIVATE "${ROOT}/engine/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
 add_test(NAME ${target} COMMAND ${target})
endforeach()
set(FXC "C:/Program Files (x86)/Windows Kits/10/bin/10.0.26100.0/x64/fxc.exe" CACHE FILEPATH "FXC compiler")
add_custom_target(WaterWorldShader ALL COMMAND "${FXC}" /nologo /T ps_5_0 /E WorldPixelMain /Fo "${CMAKE_CURRENT_BINARY_DIR}/WaterWorldPixel.cso" "${SHADER}" VERBATIM)
file(READ "${ROOT}/engine/native/src/D3D12Renderer.cpp" RENDERER_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/engine/native/src/D3D12Renderer.cpp")
string(FIND "${RENDERER_SOURCE}" "struct WaterAerialEnvironmentSelection" PARSER_BEGIN)
string(FIND "${RENDERER_SOURCE}" "\n}\nvoid D3D12Renderer::Initialize" PARSER_END)
math(EXPR PARSER_LENGTH "${PARSER_END}-${PARSER_BEGIN}")
string(SUBSTRING "${RENDERER_SOURCE}" ${PARSER_BEGIN} ${PARSER_LENGTH} PARSER)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WaterEnvironment.inc" "${PARSER}")
add_executable(WaterEnvironmentCpu WaterEnvironmentCpu.cpp)
target_compile_features(WaterEnvironmentCpu PRIVATE cxx_std_20)
target_include_directories(WaterEnvironmentCpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME WaterEnvironmentCpu COMMAND WaterEnvironmentCpu)
file(READ "${ROOT}/games/gameplay_lab/native/src/GameplayLabGame.cpp" GAME_SOURCE)
string(FIND "${GAME_SOURCE}" "void CarveShowcaseRiverShore3D(" RIVER_BEGIN)
string(FIND "${GAME_SOURCE}" "Mesh3D CreateCottageFacadeMesh3D(" RIVER_END)
math(EXPR RIVER_LENGTH "${RIVER_END}-${RIVER_BEGIN}")
string(SUBSTRING "${GAME_SOURCE}" ${RIVER_BEGIN} ${RIVER_LENGTH} RIVER_SOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/RiverProductionExtract.h" "${RIVER_SOURCE}")
set(ENVIRONMENT "${ROOT}/games/gameplay_lab/native/src/GameplayShowcaseEnvironment3D.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ENVIRONMENT}")
file(READ "${ENVIRONMENT}" ENVIRONMENT_SOURCE)
string(REGEX MATCH "waterSurface_ = ([^;]+);" PRESET "${ENVIRONMENT_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WaterPreset.inc" "auto TestWaterPreset(){return ${CMAKE_MATCH_1};}\n")
add_executable(WaterFoamCpu WaterFoamCpu.cpp "${ROOT}/engine/native/src/HeightField3D.cpp")
target_compile_features(WaterFoamCpu PRIVATE cxx_std_20)
target_compile_options(WaterFoamCpu PRIVATE /utf-8 /W4)
target_compile_definitions(WaterFoamCpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(WaterFoamCpu PRIVATE "${ROOT}/engine/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME WaterFoamCpu COMMAND WaterFoamCpu)
string(FIND "${SOURCE}" "uint CloudLatticeHash(" NOISE_BEGIN)
string(FIND "${SOURCE}" "float TerrainMesoPartitionSignal(" NOISE_END)
math(EXPR NOISE_LENGTH "${NOISE_END}-${NOISE_BEGIN}")
string(SUBSTRING "${SOURCE}" ${NOISE_BEGIN} ${NOISE_LENGTH} NOISE_SOURCE)
string(FIND "${SOURCE}" "float WaterFoamCoverage3D(" FOAM_BEGIN)
string(FIND "${SOURCE}" "float CloudFbm(" FOAM_END)
math(EXPR FOAM_LENGTH "${FOAM_END}-${FOAM_BEGIN}")
string(SUBSTRING "${SOURCE}" ${FOAM_BEGIN} ${FOAM_LENGTH} FOAM_SOURCE)
string(APPEND NOISE_SOURCE "${FOAM_SOURCE}")
string(REPLACE "float2" "Vec2" NOISE_SOURCE "${NOISE_SOURCE}")
string(REPLACE "int2" "Int2" NOISE_SOURCE "${NOISE_SOURCE}")
string(REPLACE "float CloudNoise(Vec2 coordinate)\n{" "float CloudNoise(Vec2 coordinate)\n{ ++NoiseSamples;" NOISE_SOURCE "${NOISE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WaterFoamNoise.inc" "${NOISE_SOURCE}")
string(REGEX MATCH "float3 WaterLitScatter3D\\([^}]+}" SCATTER "${SOURCE}")
if(NOT SCATTER)
 string(REGEX MATCH "const float3 physicalDeepWaterColor[^;]+;[^;]+;" LEGACY_SCATTER "${SOURCE}")
 if(NOT LEGACY_SCATTER)
  message(FATAL_ERROR "Neither production nor legacy water scatter found")
 endif()
 set(SCATTER "float3 WaterLitScatter3D(float3 waterDeepColor,float3 baseColor,float3 waterVolumeIlluminance){const unsigned waterSurfaceEnabled=1;const float3 legacyDeepColor{};${LEGACY_SCATTER}return deepWaterColor;}")
endif()
string(REGEX MATCH "float3 WaterAnalyticWave3D\\([^}]+}" WAVE "${SOURCE}")
if(WAVE)
 set(HAS_WAVE 1)
else()
 set(HAS_WAVE 0)
endif()
string(REGEX MATCH "float3 WaterReflectionFallback3D\\([^}]+}" FALLBACK "${SOURCE}")
set(ANALYTIC "${SCATTER}\n${WAVE}\n${FALLBACK}")
string(REPLACE "float3" "Vec3" ANALYTIC "${ANALYTIC}")
string(REPLACE "float2" "Vec2" ANALYTIC "${ANALYTIC}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WaterAnalyticExtract.inc" "#define WATER_ANALYTIC_PRESENT ${HAS_WAVE}\n${ANALYTIC}")
add_executable(WaterAnalyticCpu WaterAnalyticCpu.cpp)
target_compile_features(WaterAnalyticCpu PRIVATE cxx_std_20)
target_compile_options(WaterAnalyticCpu PRIVATE /utf-8)
target_include_directories(WaterAnalyticCpu PRIVATE "${ROOT}/engine/native/include" "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME WaterAnalyticCpu COMMAND WaterAnalyticCpu)
