cmake_minimum_required(VERSION 3.24)
project(GpuPassTimingCpuSmoke LANGUAGES CXX)
enable_testing()
get_filename_component(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(GpuPassTimingCpuSmoke GpuPassTimingCpuSmoke.cpp)
target_compile_features(GpuPassTimingCpuSmoke PRIVATE cxx_std_20)
target_include_directories(GpuPassTimingCpuSmoke PRIVATE "${NATIVE_ROOT}/include")
target_compile_definitions(GpuPassTimingCpuSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(GpuPassTimingCpuSmoke PRIVATE /W4 /permissive- /utf-8)
add_test(NAME GpuPassTimingCpuSmoke COMMAND GpuPassTimingCpuSmoke "${NATIVE_ROOT}/src/D3D12Renderer.cpp")
# 可選的獨立 renderer 編譯驗證；只讀既有 shader headers，不重建共用目錄。
set(TIMING_SHADER_HEADERS "" CACHE PATH "Existing generated shader headers for CPU compile-only verification")
if(TIMING_SHADER_HEADERS)
  add_library(GpuPassTimingRendererCompile OBJECT "${NATIVE_ROOT}/src/D3D12Renderer.cpp")
  target_compile_features(GpuPassTimingRendererCompile PRIVATE cxx_std_20)
  target_include_directories(GpuPassTimingRendererCompile PRIVATE "${NATIVE_ROOT}/include" "${TIMING_SHADER_HEADERS}")
  target_compile_definitions(GpuPassTimingRendererCompile PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX AEVRYND_STREAMLINE_AVAILABLE=0 AEVRYND_INTERNAL_RENDERER_TEST_HOOKS=0 AEVRYND_ENGINE_VERSION="timing-cpu")
  target_compile_options(GpuPassTimingRendererCompile PRIVATE /W4 /permissive- /utf-8)
endif()
