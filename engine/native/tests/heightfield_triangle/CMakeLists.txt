cmake_minimum_required(VERSION 3.24)
project(HeightfieldTriangleCpu LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ENGINE "${ROOT}/engine/native")
set(CPU_ENGINE_LIBRARY "" CACHE FILEPATH "Read-only existing engine archive; not full-fresh-engine")
function(cpu name source)
 add_executable(${name} "${source}" "${ENGINE}/src/HeightField3D.cpp")
 target_compile_features(${name} PRIVATE cxx_std_20)
 target_compile_options(${name} PRIVATE /utf-8 /W4 /permissive-)
 target_include_directories(${name} PRIVATE "${ENGINE}/include")
 target_compile_definitions(${name} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE AEVRYND_SOURCE_DIR=L"${ROOT}" AEVRYND_TEST_ARTIFACT_DIR=L"${CMAKE_CURRENT_BINARY_DIR}/artifacts")
 if(CPU_ENGINE_LIBRARY)
 target_link_libraries(${name} PRIVATE "${CPU_ENGINE_LIBRARY}" d3d12 dxgi windowscodecs ole32 user32 gdi32 psapi xaudio2 xinput9_1_0 bcrypt)
 endif()
 add_test(NAME ${name} COMMAND ${name})
endfunction()
cpu(HeightfieldTriangleCpu "${CMAKE_CURRENT_SOURCE_DIR}/HeightfieldTriangleCpu.cpp")
foreach(pair IN ITEMS "HeightField3D|HeightField3DSmoke" "CharacterController|CharacterControllerSmoke" "FootPlacement3D|FootPlacement3DSmoke" "TerrainClipmap3D|TerrainClipmap3DSmoke" "WorldPartitionTerrainVegetationAdapterFake|WorldPartitionTerrainVegetationAdapterFakeSmoke" "WorldPartitionTerrainVegetationDeltaComposeFake|WorldPartitionTerrainVegetationDeltaComposeFakeSmoke")
 string(REPLACE "|" ";" fields "${pair}")
 list(GET fields 0 name)
 list(GET fields 1 file)
 cpu(${name} "${ENGINE}/tests/${file}.cpp")
endforeach()
