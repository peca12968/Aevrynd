cmake_minimum_required(VERSION 3.24)
project(CloudFifth LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
file(READ "${ROOT}/engine/native/shaders/VolumetricAtmosphere.hlsl" SOURCE)
string(FIND "${SOURCE}" "uint HashLattice" START)
string(FIND "${SOURCE}" "float BlueR2Jitter" END)
math(EXPR LENGTH "${END}-${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} EXTRACT)
string(REPLACE "[unroll]" "" EXTRACT "${EXTRACT}")
string(REPLACE "[loop]" "" EXTRACT "${EXTRACT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CloudDensity.inc" "${EXTRACT}")
add_executable(CloudFifth Scan.cpp)
target_compile_features(CloudFifth PRIVATE cxx_std_20)
target_compile_options(CloudFifth PRIVATE /O2 /utf-8)
target_include_directories(CloudFifth PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME CloudFifth COMMAND CloudFifth)
