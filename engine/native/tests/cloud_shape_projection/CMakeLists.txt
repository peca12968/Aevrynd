cmake_minimum_required(VERSION 3.24)
project(CloudShapeProjection LANGUAGES CXX)
enable_testing()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(SHADER "${ROOT}/engine/native/shaders/VolumetricAtmosphere.hlsl")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHADER}")
file(READ "${SHADER}" SOURCE)
string(FIND "${SOURCE}" "uint HashLattice" START)
string(FIND "${SOURCE}" "float BlueR2Jitter" END)
math(EXPR LENGTH "${END}-${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} EXTRACT)
string(REPLACE "[unroll]" "" EXTRACT "${EXTRACT}")
string(REPLACE "[loop]" "" EXTRACT "${EXTRACT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CloudDensity.inc" "${EXTRACT}")
add_executable(CloudShapeProjection CloudShapeProjection.cpp)
target_compile_features(CloudShapeProjection PRIVATE cxx_std_20)
target_compile_options(CloudShapeProjection PRIVATE /O2 /utf-8)
target_include_directories(CloudShapeProjection PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME CloudShapeProjection COMMAND CloudShapeProjection)
